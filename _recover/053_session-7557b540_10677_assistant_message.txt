def parse_fasm(path):

    bels = collections.defaultdict(lambda: collections.defaultdict(list))

def init_bits_to_value(bits):
    \\"\\"\\"把 INIT 位索引列表转成 64-bit hex\\"\\"\\"
    val = 0
    for b in bits:
        val |= (1 << b)
    return val


def bel_to_unisim(bel_name):
    \\"\\"\\"BEL 名 → Unisim 类型\\"\\"\\"
    # A6LUT/B6LUT/C6LUT/D6LUT → LUT6
    # A5LUT/B5LUT/C5LUT/D5LUT → LUT5 (usually paired with LUT6)
    if bel_name.endswith('6LUT'):
        return 'LUT6'
    if bel_name.endswith('5LUT'):
        return 'LUT5'
    return None
